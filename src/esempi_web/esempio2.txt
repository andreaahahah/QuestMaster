        let currentNode = 'inizio';
        let gameHistory = [];
        let discoveredClues = [];

        // Funzione per caricare un nodo
        function loadNode(nodeKey) {
            console.log(' Investigando:', nodeKey);

            // Verifica se il nodo esiste
            if (!storyData[nodeKey]) {
                document.getElementById('story-text').innerHTML = '<strong> ERRORE:</strong> Traccia "' + nodeKey + '" perduta!';
                return;
            }

            const node = storyData[nodeKey];
            currentNode = nodeKey;

            // Aggiungi alla cronologia delle indagini
            gameHistory.push(nodeKey);

            // Mostra il testo con animazione
            const storyTextElement = document.getElementById('story-text');
            storyTextElement.classList.remove('slide-in');

            // Formatta il testo
            const formattedText = node.testo.replace(/\r\n/g, '<br>').replace(/\n/g, '<br>');
            storyTextElement.innerHTML = formattedText;

            // Aggiungi animazione
            setTimeout(() => {
                storyTextElement.classList.add('slide-in');
            }, 50);

            // Pulisci le scelte precedenti
            clearChoices();

            // Mostra le scelte o il finale
            if (node.scelte && node.scelte.length > 0) {
                displayChoices(node.scelte);
            } else {
                showGameOver();
            }

            // Effetto sonoro simulato (per atmosfera)
            if (nodeKey === 'cantina') {
                console.log(' *Suono di passi echeggianti*');
            } else if (nodeKey === 'verita-finale') {
                console.log(' *Sospiro di sollievo*');
            }
        }

        // Funzione per mostrare le scelte
        function displayChoices(choices) {
            const container = document.getElementById('choices-container');

            choices.forEach((choice, index) => {
                const button = document.createElement('button');
                button.innerHTML = choice.testo;
                button.className = 'choice-button';

                // Aggiungi evento click
                button.onclick = function() {
                    handleChoice(choice.next);
                };

                // Aggiungi con ritardo progressivo per effetto drammatico
                setTimeout(() => {
                    button.classList.add('slide-in');
                    container.appendChild(button);
                }, index * 150);
            });
        }

        // Funzione per pulire le scelte
        function clearChoices() {
            const container = document.getElementById('choices-container');
            container.innerHTML = '';
            document.getElementById('game-over').style.display = 'none';
        }

        // Funzione per gestire una scelta
        function handleChoice(nextNode) {
            if (nextNode && storyData[nextNode]) {
                // Aggiungi indizio scoperto
                discoveredClues.push(nextNode);
                console.log(' Indizio scoperto:', nextNode);

                loadNode(nextNode);
            } else {
                console.error(' Traccia perduta:', nextNode);
            }
        }

        // Funzione per mostrare il finale
        function showGameOver() {
            setTimeout(() => {
                const gameOverElement = document.getElementById('game-over');
                gameOverElement.style.display = 'block';
                gameOverElement.classList.add('pulse');

                console.log(' Caso risolto!');
                console.log(' Indizi scoperti:', discoveredClues.length);
                console.log(' Passi dell\'indagine:', gameHistory.length);
            }, 800);
        }

        // Funzione per ricominciare l'indagine
        function restartGame() {
            gameHistory = [];
            discoveredClues = [];
            currentNode = 'inizio';
            document.getElementById('game-over').style.display = 'none';
            document.getElementById('game-over').classList.remove('pulse');
            console.log(' Nuova indagine iniziata');
            loadNode('inizio');
        }

        // Inizializza l'indagine al caricamento della pagina
        window.onload = function() {
            console.log(' Benvenuto al castello di Lord Blackwood');
            console.log(' Indagine iniziata');
            console.log(' Dati del caso:', storyData);
            loadNode('inizio');
        };

        // Gestione errori durante l'indagine
        window.onerror = function(msg, url, line, col, error) {
            console.error(' Errore durante l\'indagine:', msg, 'alla riga', line);
            return false;
        };
