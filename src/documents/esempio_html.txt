    <script>
        // 📘 Grafo della storia (può essere anche caricato da JSON esterno se vuoi)
        const storia = {
            "inizio": {
                testo: "Ti trovi all'ingresso di una foresta misteriosa.",
                scelte: [
                    { testo: "Vai a nord", next: "nord" },
                    { testo: "Apri la porta abbandonata", next: "porta" }
                ]
            },
            "nord": {
                testo: "Cammini verso nord e trovi un lago incantato.",
                scelte: [
                    { testo: "Bevi l'acqua", next: "acqua" },
                    { testo: "Torna indietro", next: "inizio" }
                ]
            },
            "porta": {
                testo: "La porta si apre cigolando e scopri un vecchio laboratorio.",
                scelte: [
                    { testo: "Esplora il laboratorio", next: "laboratorio" },
                    { testo: "Esci e torna all'inizio", next: "inizio" }
                ]
            },
            "acqua": {
                testo: "L'acqua ti dona poteri magici. Hai vinto!",
                scelte: []
            },
            "laboratorio": {
                testo: "Trovi strani appunti e una pietra brillante. Fine del viaggio.",
                scelte: []
            }
        };

        const storyDiv = document.getElementById("story");
        const choicesDiv = document.getElementById("choices");

        function mostraNodo(nodoId) {
            const nodo = storia[nodoId];
            storyDiv.textContent = nodo.testo;
            choicesDiv.innerHTML = "";

            nodo.scelte.forEach(scelta => {
                const btn = document.createElement("button");
                btn.textContent = scelta.testo;
                btn.addEventListener("click", () => mostraNodo(scelta.next));
                choicesDiv.appendChild(btn);
            });
        }

        // Avvia la storia dal nodo "inizio"
        window.onload = () => {
            mostraNodo("inizio");
        };
    </script>
